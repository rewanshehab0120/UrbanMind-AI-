import re
import numpy as np
import pandas as pd

def clean_cairo_road_network(roads_df: pd.DataFrame) -> pd.DataFrame:
    # معالجة فاقد الحارات والسرعات استناداً للتصنيف الهرمي للطرق
    lanes_defaults = {'motorway': 4, 'trunk': 3, 'primary': 3, 'secondary': 2,
                      'tertiary': 2, 'residential': 1, 'service': 1, 'unclassified': 1}
    speed_defaults = {'motorway': 100.0, 'trunk': 80.0, 'primary': 60.0, 'secondary': 50.0,
                      'tertiary': 40.0, 'residential': 30.0, 'service': 20.0, 'unclassified': 30.0}

    roads_df['highway_clean'] = roads_df['highway'].fillna('unclassified').str.lower()
    roads_df['lanes_num'] = pd.to_numeric(roads_df['lanes'].astype(str).str.extract(r'(\d+)')[0], errors='coerce')
    roads_df['speed_num'] = pd.to_numeric(roads_df['maxspeed'].astype(str).str.extract(r'(\d+)')[0], errors='coerce')

    roads_df['lanes_imputed'] = roads_df.groupby('highway_clean')['lanes_num'].transform(lambda s: s.fillna(s.median()))
    roads_df['lanes_imputed'] = roads_df.apply(
        lambda r: lanes_defaults.get(r['highway_clean'], 1) if pd.isna(r['lanes_imputed']) else r['lanes_imputed'], axis=1
    ).astype(int)

    roads_df['speed_imputed'] = roads_df.groupby('highway_clean')['speed_num'].transform(lambda s: s.fillna(s.median()))
    roads_df['speed_imputed'] = roads_df.apply(
        lambda r: speed_defaults.get(r['highway_clean'], 40.0) if pd.isna(r['speed_imputed']) else r['speed_imputed'], axis=1
    )
    return roads_df

def normalize_arabic_districts(name: str) -> str:
    # توحيد مسميات الأحياء الـ 43 لضمان التطابق التام
    if not isinstance(name, str): return ""
    name = re.sub(r'[\u064B-\u065F\u0640]', '', name.strip())
    name = re.sub(r'[إأآا]', 'ا', name)
    name = re.sub(r'[ىي]$', 'ي', name)
    name = re.sub(r'ة$', 'ه', name)
    return name

  • العنقود 2 (الأحياء الراقية والمتكاملة): وفرة في الغطاء النباتي، تغطية عالية بالنظافة، كثافة سكانية منخفضة إلى متوسطة (المعادي، النزهة، مصر الجديدة).
  • العنقود 3 (المراكز الإدارية والتجارية): كثافة رحلات نقل فائقة، أزمنة تقاطر قصيرة، مساحات خضراء محدودة (وسط البلد، الأزبكية).
  • العنقود 4 (أحياء الامتداد والتوسع العمراني): كثافة منخفضة مع فجوات في ربط منظومة المخلفات والنقل الداخلي (طرة، التبين، بدر).

(ب) التنبؤ بالنمو السكاني حتى عام 2030 (Logistic Growth Model)

تطبيق نموذج النمو اللوجستي المشبع (Verhulst Model) بالاعتماد على معدلات الزيادة الطبيعية وسعة الاستيعاب القصوى للكثافة البنائية ($K$):

P(t) = K 1 + [ (K - P0) / P0 ] × e-r × t
Demographic Projection Engine — Python 3.11 Forecasting 
def project_cairo_2030(p0_2025: float, r_growth: float, carrying_capacity_k: float, years_horizon: int = 5):
    # حساب التعداد المستقبلي مع تثبيط التشبع العمراني للأحياء المركزية
    projections = {}
    for t in range(1, years_horizon + 1):
        year = 2025 + t
        p_t = carrying_capacity_k / (1 + ((carrying_capacity_k - p0_2025) / p0_2025) * np.exp(-r_growth * t))
        projections[year] = int(p_t)
    return projections

from ultralytics import YOLO
from shapely.geometry import Point, Polygon

class TerminalCapacityAuditor:
    def __init__(self, model_weights: str, boundary_coords: list, target_quota: int):
        self.model = YOLO(model_weights)
        self.polygon = Polygon(boundary_coords)
        self.quota = target_quota

    def audit_frame(self, frame):
        # تتبع المركبات وحساب التواجد داخل حدود الموقف
        results = self.model.track(frame, persist=True, verbose=False)
        inside_count = 0
        if results[0].boxes and results[0].boxes.id is not None:
            boxes = results[0].boxes.xyxy.cpu().numpy()
            for b in boxes:
                base_pt = Point((b[0] + b[2]) / 2.0, b[3])
                if self.polygon.contains(base_pt):
                    inside_count += 1
        
        compliance_pct = (inside_count / self.quota) * 100.0 if self.quota > 0 else 0.0
        return {"active_vehicles": inside_count, "compliance_pct": compliance_pct}
